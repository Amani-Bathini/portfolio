import numpy as np
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

class HybridKnowledgeRetriever:
    def __init__(self, semantic_chunks):
        self.chunks = semantic_chunks
        self.vectorizer = TfidfVectorizer(ngram_range=(1, 2), stop_words='english')
        self.tfidf_matrix = self.vectorizer.fit_transform([c['text'] for c in self.chunks])

    def query(self, user_query: str, alpha: float = 0.5, top_k: int = 3):
        # 1. Lexical Scoring via TF-IDF
        query_tfidf = self.vectorizer.transform([user_query])
        lexical_scores = cosine_similarity(query_tfidf, self.tfidf_matrix).flatten()

        # 2. Dense Semantic Scoring (Cosine Vector Similarity)
        query_dense = get_embeddings(user_query)
        dense_scores = np.dot([c['embedding'] for c in self.chunks], query_dense)

        # 3. Hybrid Ensembled Ranking
        final_scores = (alpha * lexical_scores) + ((1 - alpha) * dense_scores)
        top_indices = np.argsort(final_scores)[::-1][:top_k]

        return [{'chunk': self.chunks[i], 'score': float(final_scores[i])} for i in top_indices]

sql — mttr_sla_analytics.sql 
WITH AgentResolutionMetrics AS (
    SELECT
        t.assigned_agent_id,
        u.full_name AS agent_name,
        d.department_name,
        TIMESTAMPDIFF(MINUTE, t.created_at, t.resolved_at) AS resolution_minutes,
        CASE
            WHEN TIMESTAMPDIFF(MINUTE, t.created_at, t.resolved_at) <= s.max_minutes_threshold
            THEN 1 ELSE 0
        END AS is_sla_compliant,
        ROW_NUMBER() OVER(PARTITION BY t.assigned_agent_id ORDER BY t.resolved_at DESC) as recent_order
    FROM tickets t
    JOIN users u ON t.assigned_agent_id = u.id
    JOIN departments d ON u.department_id = d.id
    JOIN sla_policies s ON t.sla_tier_id = s.id
    WHERE t.status IN ('RESOLVED', 'CLOSED')
)
SELECT
    department_name,
    agent_name,
    ROUND(AVG(resolution_minutes), 1) AS mttr_minutes,
    ROUND((SUM(is_sla_compliant) / COUNT(*)) * 100, 2) AS sla_compliance_pct,
    COUNT(*) as total_resolved
FROM AgentResolutionMetrics
GROUP BY department_name, agent_name
HAVING total_resolved >= 5
ORDER BY sla_compliance_pct DESC;
